#include <stdio.h>
#include <stdlib.h>
#include <time.h>

int main(void)
{
    int n;
    int i = 0;
    int* a;

    printf("Array size: ");
    scanf_s("%d", &n);

    a = (int*)malloc(n * sizeof(int));

    if (a == NULL)
    {
        printf("Memory allocation error");
        return 1;
    }

    srand((unsigned int)time(NULL));

    while (i < n)
    {
        a[i] = rand() % 100;
        i++;
    }

    i = 0;
    while (i < n)
    {
        printf("%d ", a[i]);
        i++;
    }

    free(a);

    return 0;
}
