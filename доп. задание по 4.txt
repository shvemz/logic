#include <stdio.h>
#include <stdlib.h>
#include <time.h>
#include <windows.h>

int main(void)
{
    int n;
    int i, j;
    int sum = 0;
    int** a;

    SetConsoleCP(1251);
    SetConsoleOutputCP(1251);

    printf("Введите размер квадратной матрицы: ");
    scanf_s("%d", &n);

    if (n <= 0)
    {
        printf("Размер матрицы должен быть больше 0.\n");
        return 1;
    }

    a = (int**)malloc(n * sizeof(int*));

    if (a == NULL)
    {
        printf("Ошибка выделения памяти.\n");
        return 1;
    }

    i = 0;
    while (i < n)
    {
        a[i] = (int*)malloc(n * sizeof(int));

        if (a[i] == NULL)
        {
            printf("Ошибка выделения памяти.\n");

            while (i > 0)
            {
                i--;
                free(a[i]);
            }

            free(a);
            return 1;
        }

        i++;
    }

    srand((unsigned int)time(NULL));

    i = 0;
    while (i < n)
    {
        j = 0;
        while (j < n)
        {
            a[i][j] = rand() % 69 - 18;
            j++;
        }
        i++;
    }

    printf("\nМатрица:\n");

    i = 0;
    while (i < n)
    {
        j = 0;
        while (j < n)
        {
            printf("%d\t", a[i][j]);
            j++;
        }
        printf("\n");
        i++;
    }

    i = 0;
    while (i < n)
    {
        sum = sum + a[i][i];
        i++;
    }

    printf("\nСумма элементов главной диагонали = %d\n", sum);

    i = 0;
    while (i < n)
    {
        free(a[i]);
        i++;
    }

    free(a);

    return 0;
}
