#include <stdio.h>
#include <stdlib.h>
#include <time.h>

#define N 2000

static int a[N][N];
static int b[N][N];
static int c[N][N];

int main(void)
{
    int i, j, r;

    srand((unsigned)time(NULL));

    for (i = 0; i < N; i++)
    {
        for (j = 0; j < N; j++)
        {
            a[i][j] = rand() % 100 + 1;
            b[i][j] = rand() % 100 + 1;
            c[i][j] = 0;
        }
    }

    clock_t start = clock();

    for (i = 0; i < N; i++)
    {
        for (r = 0; r < N; r++)
        {
            int t = a[i][r];

            for (j = 0; j < N; j++)
            {
                c[i][j] += t * b[r][j];
            }
        }
    }

    clock_t end = clock();

    printf("N = %d\n", N);

    printf("time: %.6f seconds\n",
        (double)(end - start) / CLOCKS_PER_SEC);

    return 0;
}
