
#define _CRT_SECURE_NO_WARNINGS

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct node
{
    char data[256];
    int priority;
    struct node* next;
};

struct node* head = NULL;

void addElement()
{
    struct node* newNode;
    struct node* current;

    newNode = (struct node*)malloc(sizeof(struct node));

    if (newNode == NULL)
    {
        printf("Memory allocation error.\n");
        return;
    }

    printf("Enter element: ");
    scanf("%255s", newNode->data);

    printf("Enter priority: ");
    scanf("%d", &newNode->priority);

    newNode->next = NULL;

    if (head == NULL || newNode->priority > head->priority)
    {
        newNode->next = head;
        head = newNode;
    }
    else
    {
        current = head;

        while (current->next != NULL &&
            current->next->priority >= newNode->priority)
        {
            current = current->next;
        }

        newNode->next = current->next;
        current->next = newNode;
    }

    printf("Element added successfully.\n");
}

void deleteElements()
{
    int number;
    int removed = 0;
    struct node* temp;

    if (head == NULL)
    {
        printf("List is empty.\n");
        return;
    }

    printf("How many elements to delete: ");
    scanf("%d", &number);

    if (number <= 0)
    {
        printf("Invalid number.\n");
        return;
    }

    while (head != NULL && removed < number)
    {
        temp = head;
        head = head->next;
        free(temp);
        removed++;
    }

    printf("%d element(s) deleted.\n", removed);
}

/* Display the list */
void displayList()
{
    struct node* current = head;

    if (head == NULL)
    {
        printf("List is empty.\n");
        return;
    }

    printf("\nCurrent list:\n");

    while (current != NULL)
    {
        printf("Element: %s | Priority: %d\n",
            current->data,
            current->priority);

        current = current->next;
    }
}

/* Change priority of all elements with the specified name */
void changePriority()
{
    char name[256];
    int newPriority;
    int found = 0;

    struct node* current;
    struct node* nextNode;
    struct node* sorted = NULL;

    if (head == NULL)
    {
        printf("List is empty.\n");
        return;
    }

    printf("Enter element name: ");
    scanf("%255s", name);

    printf("Enter new priority: ");
    scanf("%d", &newPriority);

    /* Change priority for every element with this name */
    current = head;

    while (current != NULL)
    {
        if (strcmp(current->data, name) == 0)
        {
            current->priority = newPriority;
            found++;
        }

        current = current->next;
    }

    if (found == 0)
    {
        printf("Element not found.\n");
        return;
    }

    /* Re-sort the list by descending priority */
    current = head;
    head = NULL;

    while (current != NULL)
    {
        nextNode = current->next;

        if (sorted == NULL ||
            current->priority > sorted->priority)
        {
            current->next = sorted;
            sorted = current;
        }
        else
        {
            struct node* temp = sorted;

            while (temp->next != NULL &&
                temp->next->priority >= current->priority)
            {
                temp = temp->next;
            }

            current->next = temp->next;
            temp->next = current;
        }

        current = nextNode;
    }

    head = sorted;

    printf("Priority changed successfully for %d element(s).\n",
        found);
}

void freeList()
{
    struct node* temp;

    while (head != NULL)
    {
        temp = head;
        head = head->next;
        free(temp);
    }
}

int main()
{
    int choice;

    do
    {
        printf("\n");
        printf("=========================\n");
        printf("1. Add element\n");
        printf("2. Delete elements\n");
        printf("3. Display list\n");
        printf("4. Change priority\n");
        printf("5. Exit\n");
        printf("=========================\n");
        printf("Choose an option: ");

        scanf("%d", &choice);

        switch (choice)
        {
        case 1:
            addElement();
            break;

        case 2:
            deleteElements();
            break;

        case 3:
            displayList();
            break;

        case 4:
            changePriority();
            break;

        case 5:
            freeList();
            printf("Program finished.\n");
            break;

        default:
            printf("Invalid option.\n");
        }

    } while (choice != 5);

    return 0;
}
